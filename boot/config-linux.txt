# config-linux.txt - 리눅스를 부팅할 때의 GPU 설정.
#
# 베어메탈 펌웨어용 config.txt 와는 별개다.
# tools/mksdcard.sh --linux 가 이 파일을 config.txt 로 복사한다.

# ── 필수 ────────────────────────────────────────────────────────
arm_64bit=1

# 우리가 빌드한 커널. config.mk 의 LINUX_IMAGE 와 일치해야 한다.
kernel=test_a_123_LPzero2W_linux.img

# Zero 2 W 용 디바이스 트리.
# start.elf 가 이걸 읽어 메모리 크기·시리얼 번호 등을 패치한 뒤
# 커널에 넘긴다. 그래서 QEMU 에서는 같은 방식이 통하지 않는다.
device_tree=bcm2710-rpi-zero-2-w.dtb

# ── 시리얼 콘솔 ─────────────────────────────────────────────────
enable_uart=1

# ★ 중요: Zero 2 W 는 PL011(ttyAMA0)이 기본적으로 블루투스에 물려 있고
# 40핀 헤더의 GPIO14/15 에는 미니 UART(ttyS0)가 나온다.
# 우리 커널은 미니 UART 드라이버(8250)를 껐으므로, 이 오버레이로
# 블루투스를 떼어내 PL011 을 헤더 핀으로 돌려야 콘솔이 나온다.
# 이게 없으면 부팅은 되는데 시리얼에 아무것도 안 보인다.
dtoverlay=disable-bt

# ── USB (유선 네트워크) ─────────────────────────────────────────
# Zero 2 W 에는 이더넷 포트가 없다. micro USB 로 대신한다.
#
# dr_mode=otg 이면 케이블에 따라 자동으로 정해진다:
#   OTG 젠더 + USB 이더넷 어댑터  -> 호스트 모드 (진짜 랜선)
#   PC 에 그냥 연결               -> 가젯 모드 (PC 와 USB 네트워크)
#
# 자동 판별이 안 되면 dr_mode 를 host 나 peripheral 로 고정한다.
# ★ 반드시 "USB" 라고 인쇄된 포트에 꽂아야 한다. PWR IN 은 전원 전용이다.
dtoverlay=dwc2,dr_mode=otg

# ── 부팅 시간 ───────────────────────────────────────────────────
disable_splash=1
boot_delay=0

# ── 메모리 분할 ─────────────────────────────────────────────────
# 프레임버퍼를 아직 안 쓰므로 최소값. 화면을 켤 때 64 로 올린다.
gpu_mem=16
